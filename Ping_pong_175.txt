local dragging, dragInput, dragStart, startPos

MainFrame.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        dragging = true
        dragStart = input.Position
        startPos = MainFrame.Position
        input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then dragging = false end
        end)
    end
end)

MainFrame.InputChanged:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        dragInput = input
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if input == dragInput and dragging then
        local delta = input.Position - dragStart
        MainFrame.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

local function createToggle(labelText, yPos)
    local row = Instance.new('Frame', Content)
    row.Size = UDim2.new(1, -10, 0, 26)
    row.Position = UDim2.new(0, 5, 0, yPos)
    row.BackgroundTransparency = 1
    
    local label = Instance.new('TextLabel', row)
    label.Size = UDim2.new(0.6, 0, 1, 0)
    label.Text = labelText
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.BackgroundTransparency = 1
    label.Font = Enum.Font.Gotham
    label.TextSize = 11
    label.TextXAlignment = Enum.TextXAlignment.Left
    
    local toggle = Instance.new('Frame', row)
    toggle.Size = UDim2.new(0, 34, 0, 18)
    toggle.Position = UDim2.new(1, -38, 0.5, -9)
    toggle.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    toggle.BorderSizePixel = 0
    
    local toggleCorner = Instance.new('UICorner', toggle)
    toggleCorner.CornerRadius = UDim.new(1, 0)
    
    local knob = Instance.new('Frame', toggle)
    knob.Size = UDim2.new(0, 14, 0, 14)
    knob.Position = UDim2.new(0, 2, 0.5, -7)
    knob.BackgroundColor3 = Color3.fromRGB(200, 55, 55)
    knob.BorderSizePixel = 0
    
    local knobCorner = Instance.new('UICorner', knob)
    knobCorner.CornerRadius = UDim.new(1, 0)
    
    local btn = Instance.new('TextButton', toggle)
    btn.Size = UDim2.new(1, 0, 1, 0)
    btn.BackgroundTransparency = 1
    btn.Text = ''
    
    local state = false
    
    local function update()
        if state then
            toggle.BackgroundColor3 = Color3.fromRGB(65, 15, 15)
            knob:TweenPosition(UDim2.new(1, -16, 0.5, -7), 'Out', 'Quad', 0.15, true)
            knob.BackgroundColor3 = Color3.fromRGB(255, 115, 115)
        else
            toggle.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
            knob:TweenPosition(UDim2.new(0, 2, 0.5, -7), 'Out', 'Quad', 0.15, true)
            knob.BackgroundColor3 = Color3.fromRGB(200, 55, 55)
        end
    end
    
    return {
        row = row,
        toggle = toggle,
        knob = knob,
        btn = btn,
        get = function() return state end,
        set = function(v) state = v update() end,
        toggleState = function() state = not state update() return state end
    }
end

local antiLagActive = false

local function applyAntiLag()
    Lighting.GlobalShadows = false
    Lighting.Brightness = 1
    
    for _, v in pairs(Lighting:GetChildren()) do
        if v:IsA('BloomEffect') or v:IsA('BlurEffect') or v:IsA('SunRaysEffect') or v:IsA('ColorCorrectionEffect') or v:IsA('DepthOfFieldEffect') then
            v.Enabled = false
        end
    end
    
    for _, v in pairs(workspace:GetDescendants()) do
        if v:IsA('BasePart') then
            v.Material = Enum.Material.Plastic
            v.CastShadow = false
            if v:FindFirstChild('PointLight') then
                v.PointLight:Destroy()
            end
            if v:FindFirstChild('SurfaceLight') then
                v.SurfaceLight:Destroy()
            end
        elseif v:IsA('ParticleEmitter') then
            v.Enabled = false
        elseif v:IsA('Decal') then
            v.Transparency = 1
        elseif v:IsA('Fire') or v:IsA('Smoke') or v:IsA('Sparkles') then
            v.Enabled = false
        elseif v:IsA('Sound') then
            v.Volume = 0
        end
    end
end

local function revertAntiLag()
    Lighting.GlobalShadows = true
    Lighting.Brightness = 2
end

workspace.DescendantAdded:Connect(function(v)
    if antiLagActive then
        if v:IsA('BasePart') then
            v.Material = Enum.Material.Plastic
            v.CastShadow = false
            if v:FindFirstChild('PointLight') then
                v.PointLight:Destroy()
            end
            if v:FindFirstChild('SurfaceLight') then
                v.SurfaceLight:Destroy()
            end
        elseif v:IsA('ParticleEmitter') then
            v.Enabled = false
        elseif v:IsA('Decal') then
            v.Transparency = 1
        elseif v:IsA('Fire') or v:IsA('Smoke') or v:IsA('Sparkles') then
            v.Enabled = false
        elseif v:IsA('Sound') then
            v.Volume = 0
        end
    end
end)

antiLagToggle.btn.MouseButton1Click:Connect(function()
    antiLagActive = not antiLagActive
    antiLagToggle.toggleState()
    if antiLagActive then
        applyAntiLag()
    else
        revertAntiLag()
    end
end)

local sellCooldown = 0
local buyCooldown = 0

RunService.Heartbeat:Connect(function()
    if sellToggle.get() then
        local now = tick()
        if now - sellCooldown < 0.3 then return end
        sellCooldown = now
        
        local plots = workspace:FindFirstChild('Plots')
        if plots then
            for _, plot in ipairs(plots:GetChildren()) do
                local descendants = plot:GetDescendants()
                for _, descendant in ipairs(descendants) do
                    if descendant:IsA('ProximityPrompt') then
                        if descendant.Enabled and descendant.ActionText then
                            local text = string.lower(descendant.ActionText)
                            if string.find(text, 'sell') then
                                task.spawn(function()
                                    fireproximityprompt(descendant, 10000)
                                end)
                            end
                        end
                    end
                end
            end
        end
    end
end)

local function activatePrompt(prompt)
    if not prompt or not prompt.Enabled then return end
    pcall(function()
        prompt.HoldDuration = 0
        prompt:InputHoldBegin()
        task.wait(0.05)
        prompt:InputHoldEnd()
    end)
end

    if buyToggle.get() then
        local now = tick()
        if now - buyCooldown < 0.3 then return end
        buyCooldown = now
        
        for _, prompt in pairs(workspace:GetDescendants()) do

                local text = string.lower(prompt.ActionText)
                if string.find(text, 'purchase') or string.find(text, 'buy') or string.find(text, 'comprar') or string.find(text, 'shop') then
